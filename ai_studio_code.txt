  'src/js/audio.js': `import { Howler } from 'howler';

class AudioManager {
  constructor() {
    this.muted = false;
    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
  }

  playSynthTone(freq = 440, type = 'sine', duration = 0.3) {
    if (this.muted || !this.ctx) return;
    try {
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
      gain.gain.setValueAtTime(0.08, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + duration);
      osc.connect(gain);
      gain.connect(this.ctx.destination);
      osc.start();
      osc.stop(this.ctx.currentTime + duration);
    } catch (e) {
      console.warn("AudioContext blocked");
    }
  }

  playPortalHum() { this.playSynthTone(150, 'sawtooth', 0.8); }
  playClick() { this.playSynthTone(800, 'sine', 0.1); }
  playMagicSparkle() {
    this.playSynthTone(1200, 'sine', 0.15);
    setTimeout(() => this.playSynthTone(1600, 'sine', 0.15), 80);
  }

  toggleMute() {
    this.muted = !this.muted;
    Howler.mute(this.muted);
    return this.muted;
  }
}

export const audioManager = new AudioManager();`,

  'src/js/particles.js': `export class ParticleEngine {
  constructor(canvasId) {
    this.canvas = document.getElementById(canvasId);
    this.ctx = this.canvas.getContext('2d');
    this.particles = [];
    this.theme = 'default';
    this.resize();

    window.addEventListener('resize', () => this.resize());
    this.animate = this.animate.bind(this);
    this.initParticles();
    requestAnimationFrame(this.animate);
  }

  resize() {
    this.canvas.width = window.innerWidth;
    this.canvas.height = window.innerHeight;
  }

  setTheme(theme) {
    this.theme = theme;
    this.initParticles();
  }

  initParticles() {
    this.particles = [];
    const count = window.innerWidth < 768 ? 40 : 90;
    for (let i = 0; i < count; i++) {
      this.particles.push({
        x: Math.random() * this.canvas.width,
        y: Math.random() * this.canvas.height,
        size: Math.random() * 2.5 + 0.5,
        speedX: (Math.random() - 0.5) * 0.6,
        speedY: (Math.random() - 0.5) * 0.6,
        alpha: Math.random() * 0.7 + 0.3,
        pulse: Math.random() * 0.02
      });
    }
  }

  getColor() {
    switch (this.theme) {
      case 'web': return '#ff0055';
      case 'arcane': return '#9d4edd';
      case 'lab': return '#00f0ff';
      case 'legacy': return '#ffaa00';
      case 'illusion': return '#00ff88';
      default: return '#ffffff';
    }
  }

  animate() {
    this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
    const color = this.getColor();
    this.particles.forEach(p => {
      p.x += p.speedX;
      p.y += p.speedY;
      if (p.x < 0) p.x = this.canvas.width;
      if (p.x > this.canvas.width) p.x = 0;
      if (p.y < 0) p.y = this.canvas.height;
      if (p.y > this.canvas.height) p.y = 0;

      p.alpha += p.pulse;
      if (p.alpha > 0.9 || p.alpha < 0.2) p.pulse = -p.pulse;

      this.ctx.beginPath();
      this.ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
      this.ctx.fillStyle = color;
      this.ctx.globalAlpha = Math.max(0.1, Math.min(1, p.alpha));
      this.ctx.shadowBlur = 10;
      this.ctx.shadowColor = color;
      this.ctx.fill();
    });
    requestAnimationFrame(this.animate);
  }
}`,

import { renderAudioControls } from './components/AudioControls.js';
import { renderOpeningScene } from './scenes/OpeningScene.js';
import { renderHubScene } from './scenes/HubScene.js';
import { renderWebRealm } from './scenes/WebRealm.js';
import { renderArcaneLibrary } from './scenes/ArcaneLibrary.js';
import { renderInnovationLab } from './scenes/InnovationLab.js';
import { renderLegacyHall } from './scenes/LegacyHall.js';
import { renderIllusionGallery } from './scenes/IllusionGallery.js';
import { renderFinalPortal } from './scenes/FinalPortal.js';

class MemoryBoxApp {
  constructor() {
    this.particleEngine = new ParticleEngine('bg-canvas');
    renderAudioControls(document.getElementById('audio-control'));
    this.initRouter();
  }

  showScene(sceneId) {
    document.querySelectorAll('.scene').forEach(s => {
      s.classList.remove('active');
      s.classList.add('hidden');
    });
    const target = document.getElementById(sceneId);
    if (target) {
      target.classList.remove('hidden');
      target.classList.add('active');
    }
  }

  initRouter() {
    const openingEl = document.getElementById('scene-opening');
    const hubEl = document.getElementById('scene-hub');

    const goToHub = () => {
      this.particleEngine.setTheme('default');
      renderHubScene(hubEl, (realmId) => {
        this.particleEngine.setTheme(realmId.split('-')[0]);
        if (realmId === 'web-realm') renderWebRealm(document.getElementById('scene-web-realm'), goToHub);
        if (realmId === 'arcane-library') renderArcaneLibrary(document.getElementById('scene-arcane-library'), goToHub);
        if (realmId === 'innovation-lab') renderInnovationLab(document.getElementById('scene-innovation-lab'), goToHub);
        if (realmId === 'legacy-hall') renderLegacyHall(document.getElementById('scene-legacy-hall'), goToHub);
        if (realmId === 'illusion-gallery') renderIllusionGallery(document.getElementById('scene-illusion-gallery'), goToHub);
        this.showScene(\`scene-\${realmId}\`);
      }, () => {
        renderFinalPortal(document.getElementById('scene-final-portal'), goToHub);
        this.showScene('scene-final-portal');
      });
      this.showScene('scene-hub');
    };

    renderOpeningScene(openingEl, goToHub);
  }
}

window.addEventListener('DOMContentLoaded', () => { new MemoryBoxApp(); });`
